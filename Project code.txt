git checkout --detach cf6a440f56f1a5afad3fe49a2c3cc5895a2d4d92
./configure --minimum --without-ldap --prefix=/usr/local tac_plus

BUILD

    authorization log = /var/log/tac_plus.author
    accounting log = /var/log/tac_plus.acct

        default service = permit
        service = shell { default command = permit set priv-lvl = 15 }

        default service = permit
        enable = deny
        service = shell {
            default command = permit
            set priv-lvl = 1
            set cli-view-name = "NOC-VIEW"
        }

        enable = deny
